# Minimum runtime dependencies for the MAPLE-K screw-detection simulation.
#
# Install with:
#   pip install -r requirements.txt
#
# Then run:
#   python simulation.py
#
# No CUDA, no RealSense SDK, no UR robot libraries, no YOLO weights required.

opencv-python>=4.5

# Distributed MAPLE-K bus - managing_system/nodes.py imports rpclpy directly.
# A redis server must be reachable on localhost:6379 at runtime.
# rpclpy does not declare dash/plotly itself, but its DashboardApp
# (run_dashboard) imports them, so we pin dash here (plotly comes with it).
rpclpy>=1.0.1
redis>=4.0
dash>=2.0
